-- Calculate the total loan amount to know the total loan portfolio

SELECT SUM(Loan_Amount) AS Total_Loan_Amount
FROM Portfolio_table;

-- Total working loan

SELECT sum(Outstanding_Amount) AS Loan_at_Risk
FROM Portfolio_table
WHERE Loan_Status IN ('Active','Delinquent','Defaulted');

-- DATA EXPLORATION 

-- checking for missing values 
SELECT Loan_ID, 
		Outstanding_Amount, 
        Loan_Status, 
        Employment_Type, Location 
FROM portfolio_table 
WHERE Loan_ID IS NULL OR Outstanding_Amount IS NULL OR Loan_Status IS NULL OR Employment_Type IS NULL OR Location IS NULL ; 

-- checking invalid data 
SELECT Loan_ID, Outstanding_Amount 
FROM portfolio_table 
WHERE Outstanding_Amount < 0;

-- Exposure Summaries 

SELECT ROUND(SUM(Outstanding_Amount),2) AS Current_Exposure, 
		ROUND(AVG(Outstanding_Amount),2) AS Average_Current_Exposure, 
        ROUND(MAX(Outstanding_Amount),2) AS Maximum_Current_Exposure, 
        ROUND(MIN(Outstanding_Amount),2) AS Minimum_Current_Exposure, 
        COUNT(Loan_ID) AS Portfolio_Transaction_Volume 
FROM portfolio_table 
WHERE Loan_Status IN ('Active','Delinquent','Defaulted');
